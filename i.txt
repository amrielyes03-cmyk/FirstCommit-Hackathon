const searchInput = document.querySelector('#search-input');
const searchForm = document.querySelector('#search-form');
const animeGrid = document.querySelector('#anime-grid');

async function searchAnime(query) {
    const url = `https://api.jikan.moe/v4/anime?q=${query}`;

    try {
        animeGrid.innerHTML = '<p class="loading">Suche läuft...</p>';
        const response = await fetch(url);

        if (!response.ok) {
            throw new Error(`Fehler: ${response.status}`);
        }

        const result = await response.json();
        
        // Hier rufen wir die Anzeige-Funktion mit den Daten auf:
        displayAnime(result.data);

    } catch (error) {
    console.error("Da ging was schief: ", error.message);

    if (error.message.includes("429")) {
        animeGrid.innerHTML = `<p class="error-msg">Du suchst zu schnell! Bitte warte 2 Sekunden...</p>`;
    } else {
        animeGrid.innerHTML = `<p class="error-msg">Fehler beim Laden der Daten. Bitte versuche es erneut.</p>`;
    }
}


searchForm.addEventListener('submit', (e) => {
    e.preventDefault();
    const query = searchInput.value.trim();

    if (query) {
        searchAnime(query);
    }
});

function displayAnime(animeList) {
    // 1. Alten Inhalt leeren
    animeGrid.innerHTML = '';

    // 2. Durch jeden Anime iterieren
    animeList.forEach(anime => {
        const cardHTML = `
            <div class="anime-card">
                <img src="${anime.images.jpg.image_url}" alt="${anime.title}">
                <h3>${anime.title}</h3>
            </div>
        `;

        // 3. String als HTML am Ende des Grids anfügen
        animeGrid.insertAdjacentHTML('beforeend', cardHTML);
    });

    }
}
